    18→export interface CharLevelStats {
    19→  damage: number
    20→  fireRate: number // detik antar serangan
    21→  range: number
    22→}
    23→
    24→export interface CharDef {
    25→  /** id karakter — hero ('ali' dll) ATAU id roster ('gen-13', 'custom-…') */
    26→  id: string
    27→  name: string
    28→  shortName: string
    29→  role: string
    30→  desc: string
    31→  funFact: string
    32→  emoji: string
    33→  color: number // warna jubah utama
    34→  accent: number // warna aksen (peci/hijab/ atribut)
    35→  skin: number
    36→  cost: number
    37→  upgradeCosts: [number, number]
    38→  attack: AttackKind
    39→  unlockWave: number
    40→  levels: [CharLevelStats, CharLevelStats, CharLevelStats]
    41→  aoeRadius?: number[] // per level
    42→  slowFactor?: number[] // 0..1 multiplier kecepatan musuh
    43→  slowDuration?: number[]
    44→  knockback?: number
    45→  stunDuration?: number
    46→  /** Misbah: [jumlahPahala, intervalDetik] per level — kotak sedekah pasif */
    47→  pahalaGen?: [number, number][]
    48→  /** P7: rarity (karakter roster) — utk aura partikel & styling */
    49→  rarity?: 'umum' | 'langka' | 'epik' | 'legendaris'
    50→}
    51→
    52→export interface EnemyDef {
    53→  id: EnemyId
    54→  name: string
    55→  emoji: string
    56→  desc: string
    57→  hp: number
    58→  speed: number
    59→  damage: number // kerusakan ke masjid saat lolos
    60→  reward: number // pahala saat dihalau
    61→  scale: number
    62→  steals?: number // tuyul: mencuri pahala saat lolos
    63→  buff?: { radius: number; speedMult: number; duration: number; interval: number }
    64→  knockResist?: number // 0..1
    65→  fleesOnHit?: boolean
    66→  isBoss?: boolean
    67→  flying?: boolean // melayang — tinggi & kebal slow
    68→  slowImmune?: boolean
    69→}
    70→
    71→export interface WaveSpawn {
    72→  type: EnemyId
    73→  count: number
    74→  interval: number // detik antar spawn
    75→  delay?: number // jeda awal sekelompok spawn
    76→}
    77→
    78→export interface WaveDef {
    79→  spawns: WaveSpawn[]
    80→  reward: number
    81→  isBoss?: boolean
    82→}

   856→/** kunci pekan ISO "YYYY-Www" (Senin sebagai awal pekan). */
   857→export function weeklyKey(d = new Date()): string {
   858→  const date = new Date(Date.UTC(d.getFullYear(), d.getMonth(), d.getDate()))
   859→  const dayNum = date.getUTCDay() || 7
   860→  date.setUTCDate(date.getUTCDate() + 4 - dayNum)
   861→  const yearStart = new Date(Date.UTC(date.getUTCFullYear(), 0, 1))
   862→  const weekNo = Math.ceil(((date.getTime() - yearStart.getTime()) / 86400000 + 1) / 7)
   863→  return `${date.getUTCFullYear()}-W${String(weekNo).padStart(2, '0')}`
   864→}

   866→/** modifier pekan ini — deterministik dari kunci pekan (semua orang sama). */
   867→export function pickWeeklyModifier(key = weeklyKey()): WeeklyModifier {
   868→  let h = 0
   869→  for (let i = 0; i < key.length; i++) h = (h * 31 + key.charCodeAt(i)) >>> 0
   870→  return WEEKLY_MODIFIERS[h % WEEKLY_MODIFIERS.length]
   871→}

   873→/** berapa hari lagi sampai Senin pekan depan (1..7). */
   874→export function daysUntilNextWeek(): number {
   875→  const now = new Date()
   876→  return ((8 - (now.getDay() || 7)) % 7) || 7
   877→}
