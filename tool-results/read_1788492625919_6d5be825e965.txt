   343→
   344→function CharCreator({ onToast }: { onToast: (text: string, tone: 'good' | 'bad') => void }) {
   345→  const [custom, setCustom] = useState<CharCustom>(DEFAULT_CUSTOM)
   346→  const [saved, setSaved] = useState(false)
   347→  const [savedList, setSavedList] = useState<SavedCustomEntry[]>(() => listSavedCustoms())
   348→  const [customName, setCustomName] = useState('')
   349→  const power = POWERS.find((p) => p.id === custom.power)!
   350→
   351→  const set = (patch: Partial<CharCustom>) => {
   352→    setCustom((c) => ({ ...c, ...patch }))
   353→    setSaved(false)
   354→  }
   355→
   356→  const saveCustom = () => {
   357→    try {
   358→      const nama = customName.trim() || namaCustom()
   359→      saveCustomEntry(custom, nama)
   360→      audio.tada()
   361→      setSaved(true)
   362→      setSavedList(listSavedCustoms())
   363→      onToast(`Karakter "${nama}" tersimpan & siap dipasang! 🎉`, 'good')
   364→    } catch {
   365→      onToast('Gagal menyimpan 😅 coba lagi ya!', 'bad')
   366→    }
   367→  }
   368→
   369→  const loadSaved = (entry: SavedCustomEntry) => {
   370→    setCustom({
   371→      presentation: entry.presentation === 'anak-perempuan' ? 'anak-perempuan' : 'anak-laki',
   372→      robeColor: entry.robeColor,
   373→      accentColor: entry.accentColor,
   374→      skinColor: entry.skinColor,
   375→      hairColor: entry.hairColor,
   376→      accessory: entry.accessory,
   377→      expression: entry.expression,
   378→      power: entry.power,
   379→      variant: entry.variant,
   380→    })
   381→    setCustomName(entry.name || '')
   382→    setSaved(false)
   383→    audio.chime()
   384→    onToast(`Memuat "${entry.name || 'Karya Sendiri'}" — ubah lalu simpan lagi! 🎨`, 'good')
   385→  }
   386→
   387→  const removeSaved = (id: string) => {
   388→    deleteCustomEntry(id)
   389→    setSavedList(listSavedCustoms())
   390→    audio.mosqueHit()
   391→    onToast('Karakter dihapus dari koleksi 🗑️', 'good')
   392→  }
   393→
   394→  return (
   395→    <div className="min-h-0 flex-1 overflow-y-auto px-3 pb-6 sm:px-5">
   396→      <div className="mx-auto flex max-w-3xl flex-col gap-4 lg:flex-row">
   397→        {/* ---- preview 3D real-time ---- */}
   398→        <div className="flex shrink-0 flex-col items-center gap-3 lg:w-64">
   399→          <div className="h-48 w-48 rounded-3xl border-4 border-amber-300 bg-gradient-to-b from-sky-100 via-emerald-50 to-amber-50 p-1 shadow-lg">
   400→            <CustomPreview cc={custom} size={184} />
   401→          </div>
   402→          {/* nama karakter */}
   403→          <div className="flex w-full items-center gap-1.5 rounded-full border-2 border-amber-200 bg-white/90 px-3 py-1.5">
   404→            <span className="text-sm">🏷️</span>
   405→            <input
   406→              value={customName}
   407→              onChange={(e) => setCustomName(e.target.value.slice(0, 24))}
   408→              placeholder="Nama karaktermu…"
   409→              className="min-w-0 flex-1 bg-transparent text-sm font-bold text-[#4a3b20] outline-none placeholder:text-[#c4a86a]"
   410→              aria-label="Nama karakter custom"
   411→            />
   412→          </div>

   475→              {(['anak-laki', 'anak-perempuan'] as const).map((p) => (
