   318→  private setupPost() {
   319→    const size = new THREE.Vector2()
   320→    this.renderer.getSize(size)
   321→    this.composer = new EffectComposer(this.renderer)
   322→    this.composer.addPass(new RenderPass(this.scene, this.cameraCtrl.camera))
   323→    this.bloomPass = new UnrealBloomPass(
   324→      new THREE.Vector2(size.x, size.y),
   325→      0.45, // kekuatan — "berkah bersinar" lembut
   326→      0.6,
   327→      1.05, // threshold linear: hanya elemen HDR (lampu, kubah, orb) yang berkilau
   328→    )
   329→    this.composer.addPass(this.bloomPass)
   330→    const vignette = new ShaderPass(VignetteShader)
   331→    vignette.uniforms.offset.value = 1.15
   332→    vignette.uniforms.darkness.value = 0.32
   333→    this.composer.addPass(vignette)
   334→    // OutputPass wajib: tone mapping + konversi sRGB di akhir rantai
   335→    this.composer.addPass(new OutputPass())
   336→  }

   340→  applyQuality(q: Quality) {
   341→    this.quality = q
   342→    const dpr = typeof window !== 'undefined' ? window.devicePixelRatio : 1
   343→    if (q === 'low') {
   344→      this.renderer.setPixelRatio(1)
   345→      this.renderer.shadowMap.enabled = false
   346→      this.usePost = false
   347→    } else if (q === 'medium') {
   348→      this.renderer.setPixelRatio(1)
   349→      this.renderer.shadowMap.enabled = true
   350→      this.usePost = true
   351→    } else {
   352→      this.renderer.setPixelRatio(Math.min(dpr, 2))
   353→      this.renderer.shadowMap.enabled = true
   354→      this.usePost = true
   355→    }
   356→    gameStore.set((s) => ({ ...s, quality: q }))
   357→    this.resize()
   358→  }

   371→  private calcPinchDist(): number {
   372→    const pts = Array.from(this.activePointers.values())
   373→    if (pts.length < 2) return 0
   374→    return Math.hypot(pts[0].x - pts[1].x, pts[0].y - pts[1].y)
   375→  }

   390→  private onPointerDown = (e: PointerEvent) => {
   391→    if (e.button === 2) {
   392→      this.cancelPlacing()
   393→      return
   394→    }
   395→    // P2: catat pointer untuk pinch tracking
   396→    this.activePointers.set(e.pointerId, { x: e.clientX, y: e.clientY })
   397→    if (this.activePointers.size >= 2) {
   398→      // dua jari turun → mulai pinch, batalkan drag kamera
   399→      this.pinchMode = true
   400→      this.isDown = false
   401→      this.pinchDist = this.calcPinchDist()
   402→      this.cameraCtrl.stopMomentum()
   403→      return
   404→    }
   405→    this.isDown = true
   406→    this.pinchMode = false
   407→    this.moved = 0
   408→    this.downPos = { x: e.clientX, y: e.clientY }
   409→    const st = gameStore.get()
   410→    if (st.cameraMode !== 'iso') {
   411→      this.cameraCtrl.onPointerDown(e.clientX, e.clientY, e.button)
   412→      return
   413→    }
   414→    // mode iso & sedang memilih karakter → klik slot langsung tempatkan
   415→    if (st.selectedCharId) {
   416→      this.updatePointerNdc(e)
   417→      const slotIdx = this.raycastSlot()
   418→      if (slotIdx >= 0 && this.tryPlace(slotIdx, st.selectedCharId)) {
   419→        this.isDown = false
   420→        return
   421→      }
   422→    }
   423→    this.cameraCtrl.onPointerDown(e.clientX, e.clientY, e.button)
   424→  }
   425→
   426→  private onPointerMove = (e: PointerEvent) => {
   427→    // P2: perbarui posisi pointer aktif
   428→    if (this.activePointers.has(e.pointerId)) {
   429→      this.activePointers.set(e.pointerId, { x: e.clientX, y: e.clientY })
   430→    }
   431→    // P2: pinch-to-zoom (dua jari aktif)
   432→    if (this.pinchMode && this.activePointers.size >= 2) {
   433→      const d = this.calcPinchDist()
   434→      if (this.pinchDist > 0 && d > 0) {
   435→        this.cameraCtrl.applyPinch(d / this.pinchDist)
   436→      }
   437→      this.pinchDist = d
   438→      return
   439→    }
   440→    if (this.isDown) {
   441→      this.moved += Math.abs(e.clientX - this.downPos.x) + Math.abs(e.clientY - this.downPos.y)
   442→    }
   443→    const st = gameStore.get()
   444→    // hover slot saat drag / memilih
   445→    if (st.selectedCharId && st.cameraMode === 'iso' && !this.pinchMode) {
   446→      this.updatePointerNdc(e)
   447→      this.updateHoverSlot(st.selectedCharId)
   448→    }
   449→    if (this.isDown) {
   450→      this.cameraCtrl.onPointerMove(e.clientX, e.clientY)
   451→    }
   452→  }
   453→
   454→  private onPointerUp = (e: PointerEvent) => {
   455→    const st = gameStore.get()
   456→    // P2: lepas pointer dari pinch tracking
   457→    this.activePointers.delete(e.pointerId)
   458→    if (this.activePointers.size < 2) {
   459→      if (this.pinchMode && this.activePointers.size === 0) this.pinchMode = false
   460→      this.pinchDist = 0
   461→    }
   462→    const wasClick = this.isDown && this.moved < 8 && !this.pinchMode
   463→    this.isDown = false
   464→    this.cameraCtrl.onPointerUp()
   465→
   466→    if (this.pinchMode) return
   467→    if (st.cameraMode !== 'iso') return
   468→
   469→    // drag-drop: lepas di atas slot
   470→    if (st.selectedCharId) {
   471→      this.updatePointerNdc(e)
   472→      const slotIdx = this.raycastSlot()
   473→      if (slotIdx >= 0 && this.tryPlace(slotIdx, st.selectedCharId)) return
   474→      // lepas di area bukan slot → tetap terpilih (mode klik-klik)
   475→      return
   476→    }
   477→
   478→    if (wasClick) {
   479→      this.updatePointerNdc(e)
   480→      const tower = this.raycastTower()
   481→      if (tower) {
   482→        this.selectTower(tower)
   483→        return
   484→      }
   485→      // klik area kosong → batal pilih tower
   486→      if (st.selectedTower) this.deselectTower()
   487→    }
   488→  }

   490→  private onWheel = (e: WheelEvent) => {

   495→  private updatePointerNdc(e: PointerEvent) {
   496→    const rect = this.canvas.getBoundingClientRect()
   497→    this.pointerNdc.x = ((e.clientX - rect.left) / rect.width) * 2 - 1
   498→    this.pointerNdc.y = -((e.clientY - rect.top) / rect.height) * 2 + 1
   499→  }

   501→  private raycastSlot(): number {
   502→    this.raycaster.setFromCamera(this.pointerNdc, this.cameraCtrl.camera)
   503→    const pads = this.slots.map((s) => s.pad)
   504→    const hits = this.raycaster.intersectObjects(pads, false)
   505→    return hits.length > 0 ? (hits[0].object.userData.slotIndex as number) : -1
   506→  }
   507→
   508→  private raycastTower(): Tower | null {
   509→    this.raycaster.setFromCamera(this.pointerNdc, this.cameraCtrl.camera)
   510→    const groups = this.manager.towers.map((t) => t.group)
   511→    const hits = this.raycaster.intersectObjects(groups, true)
   512→    if (hits.length === 0) return null
   513→    let obj: THREE.Object3D | null = hits[0].object
   514→    while (obj && !obj.userData.towerSlot) obj = obj.parent
   515→    if (!obj) return null
